// JavaScript needed
const tip = document.getElementById('jsTip');
const anchor = document.getElementById('jsAnchor');

document.querySelectorAll('[data-pos]').forEach(b => {
  b.onclick = () => {
    tip.style.display = 'block';
    const a = anchor.getBoundingClientRect();
    const t = tip.getBoundingClientRect();
    const gap = 10, p = b.dataset.pos;
    let x, y;
    if (p === 'top')    { x = a.left + a.width/2 - t.width/2; y = a.top - t.height - gap; }
    if (p === 'bottom') { x = a.left + a.width/2 - t.width/2; y = a.bottom + gap; }
    if (p === 'left')   { x = a.left - t.width - gap; y = a.top + a.height/2 - t.height/2; }
    if (p === 'right')  { x = a.right + gap;          y = a.top + a.height/2 - t.height/2; }
    tip.style.left = x + 'px'; tip.style.top = y + 'px';
  };
});
// close on outside click + Escape
document.addEventListener('click', e => { if (!e.target.closest('[data-pos],#jsTip')) tip.style.display='none'; });
document.addEventListener('keydown', e => { if (e.key==='Escape') tip.style.display='none'; });

// dialog
jsModalBtn.onclick = () => jsDialog.showModal();
jsClose.onclick   = () => jsDialog.close();
